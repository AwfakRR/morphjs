import {
  useAnimate
} from '@morphjs/react';

function App() {
  const [scope, animate] =
    useAnimate<HTMLDivElement>();

  function move() {
    animate('button', { x: 100 });
  }

  return (
    <div ref={scope}>
      <button onClick={move}>
        Move
      </button>
    </div>
  );
}

import {
  Outlet
} from '@tanstack/react-router';
import {
  createRouteTransition
} from '@morphjs/tanstack';

const { enter } =
  createRouteTransition();

function onMount(
  node: HTMLElement | null
) {
  if (node) enter(node);
}

function RouteLayout() {
  return (
    <main ref={onMount}>
      <Outlet />
    </main>
  );
}
